-- Answer type: elevation - point based
SELECT ST_Value(rast, 1, ST_GeomFromText('[3_wkt]',4326)) AS elevation
FROM public.dem_us
WHERE ST_Intersects(rast, ST_GeomFromText('[3_wkt]',4326))
LIMIT 1;
