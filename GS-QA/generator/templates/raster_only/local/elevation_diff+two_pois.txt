-- SQL_TEMPLATE_START
-- Answer type: elevation difference
-- Local raster operation: two ST_Value lookups, one per POI coordinate anchor.
-- Returns absolute elevation difference in meters.
WITH p1 AS (
  SELECT ST_Value(d.rast, 1, ST_GeomFromText('[3_wkt]',4326)) AS elev
  FROM public.dem_us d
  WHERE ST_Intersects(d.rast, ST_GeomFromText('[3_wkt]',4326))
  LIMIT 1
),
p2 AS (
  SELECT ST_Value(d.rast, 1, ST_GeomFromText('[4_wkt]',4326)) AS elev
  FROM public.dem_us d
  WHERE ST_Intersects(d.rast, ST_GeomFromText('[4_wkt]',4326))
  LIMIT 1
)
SELECT
  ABS(p1.elev - p2.elev) AS elevation_difference_m,
  p1.elev AS poi1_elevation_m,
  p2.elev AS poi2_elevation_m
FROM p1, p2;
-- SQL_TEMPLATE_END
