-- SQL_TEMPLATE_START
-- Answer type: ruggedness_index
-- Focal raster operation: ST_TRI on a small patch around the POI computes the
-- mean absolute elevation difference between a cell and its 8 neighbors.
-- Higher values indicate more variable / rougher terrain.
WITH origin AS (
  SELECT ST_GeomFromText('[3_wkt]',4326) AS pt
),
patch AS (
  SELECT ST_Buffer(pt::geography, 100)::geometry AS g FROM origin
),
clipped_dem AS (
  SELECT ST_Union(ST_Clip(d.rast, p.g)) AS rast
  FROM public.dem_us d
  JOIN patch p ON ST_Intersects(d.rast, p.g)
),
tri_rast AS (
  SELECT ST_TRI(rast, 1) AS rast
  FROM clipped_dem
)
SELECT ST_Value(t.rast, 1, o.pt) AS tri_value
FROM tri_rast t, origin o;
-- SQL_TEMPLATE_END
