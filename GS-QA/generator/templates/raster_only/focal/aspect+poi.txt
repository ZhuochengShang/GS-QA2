-- Answer type: aspect
-- Focal raster operation: ST_Aspect on a small patch around the POI, then sample at the POI.
-- Patch is a ~100m geography buffer to keep the focal kernel correct across tile seams.
-- Aspect returned in degrees clockwise from north (0=N, 90=E, 180=S, 270=W).
-- Negative values indicate flat cells (no defined aspect).
WITH origin AS (
  SELECT ST_GeomFromText('[3_wkt]',4326) AS pt
),
patch AS (
  SELECT ST_Buffer(pt::geography, 100)::geometry AS g FROM origin
),
clipped_dem AS (
  SELECT ST_Union(ST_Clip(d.rast, p.g)) AS rast
  FROM public.dem_us d
  JOIN patch p ON ST_Intersects(d.rast, p.g)
),
aspect_rast AS (
  SELECT ST_Aspect(rast, 1, '32BF', 'DEGREES') AS rast
  FROM clipped_dem
)
SELECT ST_Value(a.rast, 1, o.pt) AS aspect_degrees
FROM aspect_rast a, origin o;
-- SQL_TEMPLATE_END
