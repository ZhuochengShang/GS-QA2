SELECT p.*, elev.z AS elevation
FROM pois p
JOIN LATERAL (
  SELECT ST_Value(d.rast, 1, p.geometry::geometry) AS z
  FROM public.dem_us d
  WHERE ST_Intersects(d.rast, p.geometry::geometry)
  LIMIT 1
) elev ON elev.z IS NOT NULL
