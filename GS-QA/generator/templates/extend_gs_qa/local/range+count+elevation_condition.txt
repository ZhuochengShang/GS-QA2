-- Answer type: count, POIs -> Points
-- [1] POI category; [2] distance (meters); [3] anchor POI; [4] max elevation threshold (meters)
WITH candidates AS (
  SELECT p.id, p.geometry
  FROM pois p
  WHERE ST_DWithin(p.geometry::geography, ST_GeomFromText('[3_wkt]',4326)::geography, [2])
    AND p.[1_type] = '[1]'
),
poi_elev AS (
  SELECT c.id, elev.z
  FROM candidates c
  JOIN LATERAL (
    SELECT ST_Value(d.rast, 1, c.geometry::geometry) AS z
    FROM public.dem_us d
    WHERE ST_Intersects(d.rast, c.geometry::geometry)
    LIMIT 1
  ) elev ON elev.z IS NOT NULL
)
SELECT COUNT(*) AS count
FROM poi_elev
WHERE z <= [4];
