-- Answer type: distance+elevation
-- [1] POI category; [2] distance (meters); [3] anchor POI; elevation/s at [1] (returned POIs) / elevation at a point 
SELECT
  p.*,
  ST_Distance(
    ST_GeomFromText('[3_wkt]',4326)::geography,
    p.geometry::geography
  ) AS distance,
  elev.z AS elevation
FROM pois p
JOIN LATERAL (
  SELECT ST_Value(d.rast, 1, p.geometry::geometry) AS z
  FROM public.dem_us d
  WHERE ST_Intersects(d.rast, p.geometry::geometry)
  LIMIT 1
) elev ON elev.z IS NOT NULL
WHERE ST_DWithin(
        p.geometry::geography,
        ST_GeomFromText('[3_wkt]',4326)::geography,
        [2]
      )
  AND [1_type] = '[1]';
